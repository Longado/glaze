export async function monthlyExport(customerId: string) {
  const invoices = await stripe.invoices.list({ customer: customerId, limit: 100 })
  const rows = invoices.data.map(i => [i.number, i.amount_paid / 100, i.status])
  return toCsv([["invoice", "amount", "status"], ...rows])
}
